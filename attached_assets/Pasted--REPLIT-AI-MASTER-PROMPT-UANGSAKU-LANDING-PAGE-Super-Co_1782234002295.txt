## 🎯 MISI UTAMA

Bangun **landing page produksi** untuk aplikasi **UangSaku** menggunakan **Next.js 14 App Router + Tailwind CSS + Framer Motion**. Landing page ini harus:

- Terasa seperti produk startup kelas dunia (sekelas Duolingo / Grab / GoTo)
- Menggunakan animasi profesional berbasis scroll, parallax, dan micro-interaction
- Memiliki background animated particles/mesh yang hidup dan elegan
- Responsif penuh: mobile-first, lalu tablet, lalu desktop
- Siap deploy ke Vercel dengan zero-config
- Dioptimalkan untuk konversi (CRO) menggunakan psikologi "Double-Trigger Conversion"

## 🌌 BACKGROUND ANIMATION SYSTEM

### Animated Particle Background (tsParticles)
Buat file `src/components/ui/ParticleBackground.tsx`:

```tsx
'use client'
import { useCallback } from 'react'
import Particles from '@tsparticles/react'
import { loadSlim } from '@tsparticles/slim'
import type { Engine } from 'tsparticles-engine'

export default function ParticleBackground() {
  const particlesInit = useCallback(async (engine: Engine) => {
    await loadSlim(engine)
  }, [])

  return (
    <Particles
      id="tsparticles"
      init={particlesInit}
      options={{
        fullScreen: false,
        background: { color: { value: 'transparent' } },
        fpsLimit: 60,
        interactivity: {
          events: {
            onHover: { enable: true, mode: 'repulse' },
            onClick: { enable: true, mode: 'push' },
          },
          modes: {
            repulse: { distance: 100, duration: 0.4 },
            push: { quantity: 3 },
          },
        },
        particles: {
          color: { value: ['#818cf8', '#34d399', '#fbbf24'] },
          links: {
            color: '#818cf8',
            distance: 120,
            enable: true,
            opacity: 0.15,
            width: 1,
          },
          move: {
            direction: 'none',
            enable: true,
            outModes: { default: 'bounce' },
            random: true,
            speed: 0.8,
            straight: false,
          },
          number: { density: { enable: true, area: 900 }, value: 60 },
          opacity: { value: { min: 0.1, max: 0.4 } },
          shape: { type: ['circle', 'triangle'] },
          size: { value: { min: 1, max: 4 } },
        },
        detectRetina: true,
      }}
      className="absolute inset-0 w-full h-full pointer-events-none z-0"
    />
  )
}
```

### Scroll Reveal Hook

// src/hooks/useScrollReveal.ts
import { useInView } from 'react-intersection-observer'

export function useScrollReveal(threshold = 0.15) {
  const { ref, inView } = useInView({
    threshold,
    triggerOnce: true,
  })
  return { ref, inView }
}
