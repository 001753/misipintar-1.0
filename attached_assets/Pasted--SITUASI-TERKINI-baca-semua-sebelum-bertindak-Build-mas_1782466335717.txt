### Fix 2: Nonaktifkan firebase-admin sepenuhnya dari build path

Buka file src/lib/fcm.ts (atau nama file firebase/FCM yang ada).
GANTI SELURUH ISI FILE dengan versi lazy + build-safe berikut:

```typescript
// fcm.ts — lazy init, aman di build worker cPanel
// firebase-admin tidak boleh di-load saat Next.js build

let _messaging: import('firebase-admin/messaging').Messaging | null = null;

async function getMessaging() {
  // Jangan load firebase-admin saat build
  if (process.env.NEXT_BUILD === '1') return null;
  if (typeof window !== 'undefined') return null;

  if (_messaging) return _messaging;

  const projectId = process.env.FIREBASE_PROJECT_ID;
  const clientEmail = process.env.FIREBASE_CLIENT_EMAIL;
  const privateKey = process.env.FIREBASE_PRIVATE_KEY;

  if (!projectId || !clientEmail || !privateKey) {
    console.log('[FCM] Firebase env tidak lengkap — push notifications dinonaktifkan');
    return null;
  }

  try {
    const { initializeApp, getApps, cert } = await import('firebase-admin/app');
    const { getMessaging: _getMessaging } = await import('firebase-admin/messaging');

    const app = getApps().length === 0
      ? initializeApp({ credential: cert({ projectId, clientEmail, privateKey: privateKey.replace(/\\n/g, '\n') }) })
      : getApps()[0];

    _messaging = _getMessaging(app);
    return _messaging;
  } catch (err) {
    console.error('[FCM] Gagal init firebase-admin:', err);
    return null;
  }
}

export async function sendPushNotification(token: string, title: string, body: string) {
  const messaging = await getMessaging();
  if (!messaging) return null;
  try {
    return await messaging.send({ token, notification: { title, body } });
  } catch (err) {
    console.error('[FCM] Gagal kirim notifikasi:', err);
    return null;
  }
}

export async function sendMulticastNotification(tokens: string[], title: string, body: string) {
  const messaging = await getMessaging();
  if (!messaging) return null;
  try {
    return await messaging.sendEachForMulticast({ tokens, notification: { title, body } });
  } catch (err) {
    console.error('[FCM] Gagal kirim multicast:', err);
    return null;
  }
}
```

### Fix 3: Cek dan perbaiki src/lib/redis.ts

Pastikan tidak ada static import ioredis di baris teratas.
File HARUS menggunakan lazy require seperti ini:

```typescript
// redis.ts — lazy init, aman di build worker
let _redis: any = null;

export function getRedis() {
  if (process.env.NEXT_BUILD === '1') return null;
  if (!process.env.REDIS_URL) {
    console.log('[redis] REDIS_URL not set — rate limiting disabled');
    return null;
  }
  if (_redis) return _redis;
  const Redis = require('ioredis');
  _redis = new Redis(process.env.REDIS_URL);
  return _redis;
}

// Backward compat jika ada yang import `redis` langsung
export const redis = new Proxy({} as any, {
  get(_t, prop) {
    const r = getRedis();
    if (!r) return async () => null;
    return typeof r[prop] === 'function' ? r[prop].bind(r) : r[prop];
  }
});
```
